# Mesa: the console's D3D12 (feature level 11_0) keeps Mesa below GL 4.3; BAR's GL4 path needs it.
MESA_GL_VERSION_OVERRIDE=4.6COMPAT
MESA_GLSL_VERSION_OVERRIDE=460
# Host: the video memory budget the game sees (Mesa evicts to stay inside it). Too low and Mesa evicts all the
# time: at 640 MB parts of BAR's frame (the top bar, units) went missing for single frames, about 27 times in
# 10 s; at 2048 none. With host-lazy.txt the app peaks at 4.2 GB with the GPU at 1.75 GB.
HOST_GPU_BUDGET_MB=2048
# Mesa: no glthread (Mesa's drirc turns it on for spring.exe). On the console it draws every in-game frame black;
# the loading screen and a debug context (which disables glthread) render, and so does this.
mesa_glthread=false
